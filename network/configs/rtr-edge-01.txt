! rtr-edge-01  (Cisco 2911)  -  AST-1004
! Role: edge router. Inter-VLAN routing (router-on-a-stick) and DHCP server
! Site: DE-KOELN-HQ, DC-Rack-1     Holds the gateway address for every VLAN
! Uplink: Gi0/0 to sw-core-01 Gi0/1, 802.1Q trunk, native VLAN 99
!
! Exported from the working lab in network/koeln-hq-lan.pkt. Not hand written.
!
! One physical interface serves five VLANs by splitting Gi0/0 into subinterfaces,
! one per VLAN. Each subinterface holds the gateway address for its VLAN, so the
! 10.0.20.1 below is what every laptop in VLAN 20 uses as its default gateway.
!
! Gi0/0.99 is the native VLAN and deliberately has no address. Native traffic
! crosses a trunk untagged and nothing is meant to route on it, so it needs none.
!
! The three excluded-address lines matter more than they look. DHCP leases out of
! 10.0.20.0/24, so anything statically assigned inside that range has to be
! excluded or the router will eventually hand it to something else. 10.0.20.80 is
! the printer (prn-2f-01, AST-1014) and was added after building the lab exposed
! the clash. It would have worked fine until the day a new device joined.
!
version 15.1
no service timestamps log datetime msec
no service timestamps debug datetime msec
no service password-encryption
!
hostname rtr-edge-01
!
ip dhcp excluded-address 10.0.20.1 10.0.20.50
ip dhcp excluded-address 10.0.40.1 10.0.40.20
ip dhcp excluded-address 10.0.20.80
!
ip dhcp pool CLIENTS
 network 10.0.20.0 255.255.255.0
 default-router 10.0.20.1
 dns-server 10.0.30.10
ip dhcp pool WIFI
 network 10.0.40.0 255.255.255.0
 default-router 10.0.40.1
 dns-server 10.0.30.10
!
ip cef
no ipv6 cef
!
license udi pid CISCO2911/K9 sn FTX152483GD-
!
spanning-tree mode pvst
!
interface GigabitEthernet0/0
 no ip address
 duplex auto
 speed auto
!
interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 10.0.10.1 255.255.255.0
!
interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 10.0.20.1 255.255.255.0
!
interface GigabitEthernet0/0.30
 encapsulation dot1Q 30
 ip address 10.0.30.1 255.255.255.0
!
interface GigabitEthernet0/0.40
 encapsulation dot1Q 40
 ip address 10.0.40.1 255.255.255.0
!
interface GigabitEthernet0/0.99
 encapsulation dot1Q 99 native
 no ip address
!
interface GigabitEthernet0/1
 no ip address
 duplex auto
 speed auto
 shutdown
!
interface GigabitEthernet0/2
 no ip address
 duplex auto
 speed auto
 shutdown
!
interface Vlan1
 no ip address
 shutdown
!
ip classless
!
ip flow-export version 9
!
line con 0
!
line aux 0
!
line vty 0 4
 login
!
end
